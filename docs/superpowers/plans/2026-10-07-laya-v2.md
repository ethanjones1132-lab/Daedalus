```python
HIDDEN_SIGNALS = ("noul", "kind_p")
IDENTITY = {"classify_wording": "v1", "verify_form": "noul", "hidden_signal": "noul",
            "platt": {q: [1.0, 0.0] for q in QUANTITIES}}


def hidden_of(raw, signal):
    """The task signal the v2 rule reads (spec v2 §3): the yes/no "unseen" probability, or the kind question's
    probability for the hidden-code option."""
    return raw["kind_p"]["unseen"] if signal == "kind_p" else raw["unseen"]


def hidden_aucs(label_rows, cats, wordings=WORDINGS):
    """(wording, signal) -> AUC for category-B tasks on the pool."""
    out = {}
    for w in wordings:
        cards = cards_of(label_rows, w)
        for s in HIDDEN_SIGNALS:
            out[(w, s)] = pairs_auc([(hidden_of(c, s), cats[t] == "B") for t, c in cards.items() if t in cats])
    return out
```

`calibrate_card(raw, platt, hidden_signal="noul")` adds `"hidden": hidden_of(raw, hidden_signal)`. In `fit_cmd`, the chosen `(wording, signal)` is the maximum of `hidden_aucs` (by `_num`). It records `"hidden_signal"`, and the report records `"hidden_auc": {"v1/kind_p": ..., ...}`. The verify form choice is unchanged.

```python
def decile_grid(values):
    """Cutoff candidates for the targeted form: the 0th..90th percentile values (one per task) plus NEVER."""
    v = sorted(values)
    return tuple(sorted({v[int(len(v) * i / 10)] for i in range(10)})) + (NEVER,)


def fit_targeted(recs, cards, budget, use_p):
    """Spec v2 §2: (rule, solved, mean secs) over c_hidden in the decile grid (and the verify thresholds)."""
    memo, best = {}, None
    thresholds = list(itertools.product(T_HI, T_LO)) if use_p else [(NEVER, -1.0)]
    for c in decile_grid([cd["hidden"] for cd in cards.values()]):
        for t_hi, t_lo in thresholds:
            rule = dict(form="targeted", c_hidden=c, t_hi=t_hi, t_lo=t_lo)
            solved, secs = system(recs, cards, rule, use_p, memo)
            if secs <= budget and (best is None or (solved, -secs) > (best[1], -best[2])):
                best = (rule, solved, secs)
    return best
```

`choose` starts with: `if rule.get("form") == "targeted": return "R" if card is None else ("P" if card["hidden"] >= rule["c_hidden"] else "R")`. `records` passes `calib.get("hidden_signal", "noul")` to `calibrate_card`. `fit_cmd` gains `--form {targeted,v1}` (default `targeted`), uses `fit_targeted` for it, and writes `"form"` and `"grid"`.

```python
def note_for(card, play, rule):
    """Spec v2 §5: under the targeted form only probed tasks get the unseen-code note; v1 rules keep per-kind notes."""
    if card is None:
        return ""
    if rule.get("form") == "targeted":
        return NOTES["unseen"] + "\n\n" if play == "P" else ""
    return NOTES[card["kind"]] + "\n\n"
```
